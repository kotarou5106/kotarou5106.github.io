{{- $page := . -}} {{- $html := "" -}} {{- $compatible := false -}} {{- with .File -}} {{/* Read the source-relative file: Hugo's bundle resource list can miss an atomically replaced JSON resource during live reload. */}} {{- $relative := strings.TrimPrefix (printf "%s/" (strings.TrimSuffix "/" hugo.WorkingDir)) .Filename -}} {{- $file := path.Join (path.Dir $relative) "blog-writer-render.json" -}} {{- if os.FileExists $file -}} {{- $copy := os.ReadFile $file | transform.Unmarshal -}} {{- $source := replace (replace $page.RawContent "\r\n" "\n") "\r" "\n" -}} {{- if and (eq (int $copy.version) 1) (eq $copy.sourceHash (crypto.SHA256 (strings.Trim $source "\n"))) (gt (len $copy.fragments) 0) -}} {{- $html = $page.RenderString (dict "display" "block") $copy.markdown -}} {{- $compatible = true -}} {{- if $copy.hasMath -}} {{- $html = printf "%s" $html | safeHTML -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if not $compatible -}} {{- $html = .Content -}} {{- end -}} {{- return $html -}}
